cmake_minimum_required(VERSION 3.22.1)
project(gyrolock_vulkan LANGUAGES CXX)
add_library(gyrolock_vulkan SHARED gyrolock_vulkan.cpp)
find_library(log_lib log)
find_library(android_lib android)
find_library(vulkan_lib vulkan)
target_compile_features(gyrolock_vulkan PRIVATE cxx_std_17)
target_compile_definitions(gyrolock_vulkan PRIVATE VK_USE_PLATFORM_ANDROID_KHR=1)
target_compile_options(gyrolock_vulkan PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(gyrolock_vulkan ${log_lib} ${android_lib} ${vulkan_lib})
